algoritmo "Estatisticas de Numeros com Valor Sentinela"

var
   numero, quantidade, soma, maior, menor: inteiro

inicio
   quantidade <- 0
   soma       <- 0
   maior      <- -2147483647
   menor      <- 2147483647

   escreva("Digite um número (0 para encerrar): ")
   leia(numero)

   enquanto numero <> 0 faca
      quantidade <- quantidade + 1
      soma <- soma + numero

      se numero > maior entao
         maior <- numero
      fimse
      se numero < menor entao
         menor <- numero
      fimse

      escreva("Digite um número (0 para encerrar): ")
      leia(numero)
   fimenquanto

   se quantidade > 0 entao
      escreval("Quantidade de números digitados: ", quantidade)
      escreval("Soma total: ", soma)
      escreval("Maior número: ", maior)
      escreval("Menor número: ", menor)
   senao
      escreval("Nenhum número foi digitado.")
   fimse

fimalgoritmo

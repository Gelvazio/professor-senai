algoritmo "Tabuada Completa"

var
   numero, i: inteiro

inicio
   faca
      escreva("Digite um número entre 1 e 10: ")
      leia(numero)
      se (numero < 1) ou (numero > 10) entao
         escreval("Número inválido! Digite entre 1 e 10.")
      fimse
   enquanto (numero < 1) ou (numero > 10)

   escreval("=== Tabuada do ", numero, " ===")
   para i de 1 ate 10 faca
      escreval(numero, " x ", i, " = ", numero * i)
   fimpara

fimalgoritmo

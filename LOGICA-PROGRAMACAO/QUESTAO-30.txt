algoritmo "Desafio Completo - Sistema Integrado de Compras"

var
   filaClientes  : vetor[1..5] de cadeia
   totaisConsumo : vetor[1..5] de real
   fimFila       : inteiro
   qtdAtendidos  : inteiro
   opcao         : inteiro

procedimento enfileirar_cliente()
inicio
   se fimFila = 5 entao
      escreval("Fila cheia! Não é possível adicionar mais clientes.")
   senao
      fimFila <- fimFila + 1
      escreva("Nome do cliente: ")
      leia(filaClientes[fimFila])
      escreval("Cliente '", filaClientes[fimFila], "' adicionado à fila.")
   fimse
fimprocedimento

procedimento atender_cliente()
var
   i: inteiro
   valor: real
inicio
   se fimFila = 0 entao
      escreval("Nenhum cliente na fila.")
   senao
      escreval("Atendendo: ", filaClientes[1])
      escreva("Valor total da compra: R$ ")
      leia(valor)
      qtdAtendidos <- qtdAtendidos + 1
      totaisConsumo[qtdAtendidos] <- valor
      escreval("Compra registrada: R$ ", valor:0:2)

      para i de 1 ate fimFila - 1 faca
         filaClientes[i] <- filaClientes[i + 1]
      fimpara
      fimFila <- fimFila - 1
   fimse
fimprocedimento

   total: real
